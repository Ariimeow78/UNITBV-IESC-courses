#+title: Curs Big Number Multiplication
#+description: tips and tricks

#+BEGIN_SRC C

uint32_t a = 987654321;
uint32_t b = 234567891;

uint32_t a[32], b[32];
uint32_t c[64];

void multiply_limbs(const uint32_t a[32], const uint32_t b[32], uint32_t c[64]) {
    // Step A: Initialize output buffer to zero
    for (int k = 0; k < 64; k++) {
        c[k] = 0;
    }

    // Step B: Nested loop over all limbs
    for (int i = 0; i < 32; i++) {
        uint64_t carry = 0; // Carry resets at the start of each row (a[i])
        
        for (int j = 0; j < 32; j++) {
            // Step C: Multiply and accumulate into 64-bit container
            uint64_t current = (uint64_t)a[i] * b[j] + c[i + j] + carry;

            // Step D: Extract bottom 32 bits into destination array
            c[i + j] = (uint32_t)(current & 0xFFFFFFFF);

            // Step E: Extract top 32 bits as carry for the next iteration
            carry = current >> 32;
        }

        // Step F: Apply remaining carry to the top slot of this row
        c[i + 32] += (uint32_t)carry;
    }
}

#+END_SRC
